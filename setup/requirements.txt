# Training-host deps. torch bundles its own CUDA runtime, but pufferlib 3.0
# compiles a torch CUDA extension against the system nvcc at install time,
# so torch's CUDA major must match nvcc. The default-index torch 2.13 is
# CUDA 13; on a CUDA 12.x host (driver < 580) install torch from the cu128
# index first, then pufferlib without build isolation, then this file:
#   uv pip install "torch<2.13" --index-url https://download.pytorch.org/whl/cu128
#   uv pip install --no-build-isolation pufferlib==3.0.0
#   sed 's/^torch==.*/torch<2.13/' setup/requirements.txt > /tmp/req_cu128.txt
#   PTCG_REQUIREMENTS=/tmp/req_cu128.txt bash setup/bootstrap.sh
